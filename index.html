// Program.cs using System; using System.Linq; using System.Threading.Tasks; using Spectre.Console; using ZuzifyQXE; class Program { static Core core = new(); static Theme T => core.T; static async Task Main() { Console.Title = "ZuzifyQXE"; await core.Load(); await Boot(); await core.Connect(); if (core.Cfg.UserId == null && !string.IsNullOrWhiteSpace(core.Cfg.Username)) { var id = await core.EnsureUser(core.Cfg.Username); if (id.HasValue) { core.Cfg.UserId = id; await core.Save(); } } await Loop(); } static async Task Boot() { AnsiConsole.Clear(); AnsiConsole.Write(new FigletText("ZuzifyQXE").Centered().Color(Color.Cyan1)); AnsiConsole.Write(new Rule($"[grey]{App.Version}[/]").Centered()); await AnsiConsole.Progress() .AutoClear(true) .Columns(new ProgressColumn[] { new TaskDescriptionColumn(), new ProgressBarColumn(), new SpinnerColumn() }) .StartAsync(async ctx => { foreach (var s in new[] { "core", "discord", "editor", "ready" }) { var t = ctx.AddTask(s); while (!t.IsFinished) { t.Increment(Random.Shared.Next(5, 12)); await Task.Delay(30); } } }); } static async Task Loop() { if (string.IsNullOrWhiteSpace(core.Cfg.Username)) { core.Cfg.Username = AnsiConsole.Ask("username:"); if (AnsiConsole.Confirm("add bot token for live lookups?", false)) core.Cfg.BotToken = AnsiConsole.Ask("bot token:"); await core.Save(); } while (true) { AnsiConsole.Clear(); AnsiConsole.Write(new Rule($"ZuzifyQXE {(core.Online ? "[green]online[/]" : "[red]offline[/]")}").RuleStyle(T.Primary)); var table = new Table().Border(TableBorder.Rounded).HideHeaders(); table.AddColumn("k").AddColumn("v"); table.AddRow("lookups", core.Cfg.TotalLookups.ToString()); table.AddRow("badges", core.Cfg.Badges.Count.ToString()); table.AddRow("theme", core.Cfg.Theme.ToString()); table.AddRow("bot", string.IsNullOrWhiteSpace(core.Cfg.BotToken) ? "no" : "yes"); AnsiConsole.Write(table); AnsiConsole.WriteLine(); var choice = AnsiConsole.Prompt(new SelectionPrompt() .AddChoices( "lookup", "favorites", "notes", "compare", "code editor", "list codes", "achievements", "theme", "settings", "export", "command", "exit" )); switch (choice) { case "lookup": await Lookup(); break; case "favorites": await Favorites(); break; case "notes": await Notes(); break; case "compare": await Compare(); break; case "code editor": await Editor(); break; case "list codes": ListCodes(); break; case "achievements": Badges(); break; case "theme": await Theme(); break; case "settings": await Settings(); break; case "export": await Export(); break; case "command": await Cmd(); break; case "exit": return; } } } static async Task Lookup() { var q = AnsiConsole.Ask("id / username:"); Profile? p = null; await AnsiConsole.Status().StartAsync("looking up...", async _ => p = await core.Lookup(q)); if (p == null) { AnsiConsole.MarkupLine($"[{T.Error}]not found[/]"); Pause(); return; } var panel = new Panel( $"id {p.DiscordId}\n" + $"username {p.Username}\n" + $"display {p.GlobalName ?? "—"}\n" + $"bot {p.Bot}\n" + $"lookups {p.LookupCount}\n" + $"updated {p.LastUpdated:u}") .Header("profile") .BorderColor(Color.Cyan1); AnsiConsole.Write(panel); if (AnsiConsole.Confirm("favorite?", false)) await core.AddFavorite(p.DiscordId); Pause(); } static async Task Favorites() { var list = await core.GetFavorites(); if (list.Count == 0) { AnsiConsole.MarkupLine("[grey]empty[/]"); Pause(); return; } var t = new Table().Border(TableBorder.Rounded).AddColumn("discord id"); foreach (var id in list) t.AddRow(id); AnsiConsole.Write(t); Pause(); } static async Task Notes() { var id = AnsiConsole.Ask("discord id:"); var text = AnsiConsole.Ask("note:"); if (string.IsNullOrWhiteSpace(id) || string.IsNullOrWhiteSpace(text)) return; await core.AddNote(id, text); AnsiConsole.MarkupLine($"[{T.Success}]saved[/]"); Pause(); } static async Task Compare() { var a = AnsiConsole.Ask("first:"); var b = AnsiConsole.Ask("second:"); Profile? p1 = null, p2 = null; await AnsiConsole.Status().StartAsync("loading...", async _ => { p1 = await core.Lookup(a); p2 = await core.Lookup(b); }); if (p1 == null || p2 == null) { AnsiConsole.MarkupLine($"[{T.Error}]failed[/]"); Pause(); return; } var t = new Table().Border(TableBorder.Rounded); t.AddColumn("field").AddColumn(p1.Username ?? "A").AddColumn(p2.Username ?? "B"); t.AddRow("id", p1.DiscordId, p2.DiscordId); t.AddRow("display", p1.GlobalName ?? "—", p2.GlobalName ?? "—"); t.AddRow("bot", p1.Bot.ToString(), p2.Bot.ToString()); t.AddRow("lookups", p1.LookupCount.ToString(), p2.LookupCount.ToString()); AnsiConsole.Write(t); await core.Unlock("compare"); Pause(); } static async Task Editor() { var name = AnsiConsole.Ask("filename (e.g. test.cs):"); if (string.IsNullOrWhiteSpace(name)) return; var existing = await core.LoadCode(name); if (existing != null) AnsiConsole.MarkupLine($"[grey]loaded existing ({existing.Length} chars)[/]"); AnsiConsole.MarkupLine("[grey]paste code below. end with a line containing only :::END:::[/]"); var lines = new List(); while (true) { var line = Console.ReadLine(); if (line == ":::END:::") break; lines.Add(line ?? ""); } var content = string.Join("\n", lines); await core.SaveCode(name, content); AnsiConsole.MarkupLine($"[{T.Success}]saved → {name}[/]"); Pause(); } static void ListCodes() { var files = core.ListCodes(); if (files.Count == 0) { AnsiConsole.MarkupLine("[grey]no files[/]"); Pause(); return; } var t = new Table().Border(TableBorder.Rounded).AddColumn("file"); foreach (var f in files) t.AddRow(f); AnsiConsole.Write(t); Pause(); } static void Badges() { var t = new Table().Border(TableBorder.Rounded).AddColumn("badge").AddColumn("has"); var all = new Dictionary { ["first"] = "First Contact", ["scout"] = "Scout (10)", ["tracker"] = "Tracker (50)", ["archivist"] = "Archivist (200)", ["compare"] = "Compared users" }; foreach (var (k, title) in all) t.AddRow(title, core.Cfg.Badges.Contains(k) ? "[green]yes[/]" : "[grey]no[/]"); AnsiConsole.Write(t); Pause(); } static async Task Theme() { var t = AnsiConsole.Prompt(new SelectionPrompt().Title("theme").AddChoices(Enum.GetValues())); core.Cfg.Theme = t; await core.Save(); } static async Task Settings() { core.Cfg.BotToken = AnsiConsole.Ask("bot token:", core.Cfg.BotToken); await core.Save(); AnsiConsole.MarkupLine("[green]saved[/]"); Pause(); } static async Task Export() { var json = core.Export(); var path = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.Desktop), $"zqxe_{DateTime.Now:yyyyMMdd_HHmmss}.json"); await File.WriteAllTextAsync(path, json); AnsiConsole.MarkupLine($"[green]{path}[/]"); await core.Unlock("export"); Pause(); } static async Task Cmd() { var input = AnsiConsole.Ask(">").Trim().ToLowerInvariant(); if (input.StartsWith("lookup ")) { var p = await core.Lookup(input[7..]); AnsiConsole.MarkupLine(p != null ? $"{p.Username} ({p.DiscordId})" : "[red]not found[/]"); } else if (input.StartsWith("theme ")) { if (Enum.TryParse(input[6..], true, out var t)) { core.Cfg.Theme = t; await core.Save(); } } else if (input == "export") { await Export(); return; } else AnsiConsole.MarkupLine("lookup | theme | export"); Pause(); } static void Pause() { AnsiConsole.MarkupLine("\n[grey]press any key[/]"); Console.ReadKey(true); } }
